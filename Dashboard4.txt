import client from "./ApiClient";

import { ApiEndpoints }
        from "../config/ApiEndpoints";

import type {
    DashboardFilter,
    ExecutiveDashboardResponse,
    DashboardDataPoint,
    KpiValue
} from "../model/dashboard";

export default class DashboardApi {

    static async getLatestExecutiveDashboard(
        filter: DashboardFilter
    ): Promise<ExecutiveDashboardResponse> {

        const response =
            await client.post<ExecutiveDashboardResponse>(
                ApiEndpoints.dashboard.latestExecutive,
                filter
            );

        return response.data;
    }

    static async getExecutiveDashboard(
        importJobId: number,
        filter: DashboardFilter
    ): Promise<ExecutiveDashboardResponse> {

        const response =
            await client.post<ExecutiveDashboardResponse>(
                ApiEndpoints.dashboard.executive(
                    importJobId),
                filter
            );

        return response.data;
    }
