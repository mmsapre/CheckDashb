(
    'AGING_DISTRIBUTION',
    'Open vulnerability aging',
    'STACKED_BAR',
    70,
    $sql$
    INSERT INTO dashboard_result
    WITH bucketed AS
    (
        SELECT
            CASE
                WHEN found_date IS NULL THEN 'UNKNOWN'
                WHEN DATE_DIFF(
                    'day',
                    found_date,
                    CURRENT_DATE
                ) <= 30 THEN '0-30'
                WHEN DATE_DIFF(
                    'day',
                    found_date,
                    CURRENT_DATE
                ) <= 60 THEN '31-60'
                WHEN DATE_DIFF(
                    'day',
                    found_date,
                    CURRENT_DATE
                ) <= 90 THEN '61-90'
                WHEN DATE_DIFF(
                    'day',
                    found_date,
                    CURRENT_DATE
                ) <= 180 THEN '91-180'
                ELSE '180+'
            END AS age_bucket,
            COALESCE(
                UPPER(severity),
                'UNKNOWN'
            ) AS severity
        FROM stage_data
        WHERE UPPER(COALESCE(
            mitigation_status, 'OPEN'
        )) NOT IN (
            'MITIGATED',
            'RESOLVED',
            'CLOSED'
        )
    )
    SELECT
        ${IMPORT_JOB_ID},
        'AGING_DISTRIBUTION',
        age_bucket,
        severity,
        NULL,
        COUNT(*),
        NULL,
        NULL,
        NULL,
        CASE age_bucket
            WHEN '0-30' THEN 1
            WHEN '31-60' THEN 2
            WHEN '61-90' THEN 3
            WHEN '91-180' THEN 4
            WHEN '180+' THEN 5
            ELSE 6
        END,
        NULL
    FROM bucketed
    GROUP BY age_bucket, severity
    $sql$

            END AS epss_bucket,
            severity,
            kev,
            epss
        FROM stage_data
    )
    SELECT
        ${IMPORT_JOB_ID},
        'EPSS_DISTRIBUTION',
        epss_bucket,
        NULL,
        NULL,
        COUNT(*),
        COUNT(*) FILTER (
            WHERE UPPER(severity) = 'CRITICAL'
        ),
        COUNT(*) FILTER (
            WHERE kev = TRUE
        ),
        NULL,
        CASE epss_bucket
            WHEN '0.90-1.00' THEN 1
            WHEN '0.70-0.89' THEN 2
            WHEN '0.50-0.69' THEN 3
            WHEN '0.20-0.49' THEN 4
            WHEN '0.00-0.19' THEN 5
            ELSE 6
        END,
        json_object(
            'averageEpss',
            AVG(epss)
        )
    FROM bucketed
    GROUP BY epss_bucket
