export interface DashboardDataPoint {
    category: string;
    series?: string | null;
    subCategory?: string | null;
    value: number;
    secondaryValue?: number | null;
    tertiaryValue?: number | null;
    sortOrder: number;
    payload?: Record<string, unknown> | null;
}

export interface KpiValue {
    code: string;
    label: string;
    value: number;
    unit?: string | null;
    description?: string | null;
}

    importJobId: number;
    calculatedAt: string;

    jawsRatio: number;
    jawsTrendText: string;

    riskPosture: DashboardDataPoint[];

    topExposedService: KpiValue;
    overallJawsRatio: KpiValue;

    topL5Departments: DashboardDataPoint[];
    topServices: DashboardDataPoint[];

    insights: string[];
}
