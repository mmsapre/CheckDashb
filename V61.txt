CREATE TABLE dashboard_aggregation_definition
(
    aggregation_id          BIGSERIAL PRIMARY KEY,

    aggregation_code        VARCHAR(100) NOT NULL UNIQUE,

    aggregation_name        VARCHAR(200) NOT NULL,

    chart_type              VARCHAR(50) NOT NULL,

    display_order           INTEGER NOT NULL,

    enabled                 BOOLEAN NOT NULL DEFAULT TRUE,

    description             VARCHAR(1000),

    aggregation_sql         TEXT NOT NULL,

    created_at              TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,

    updated_at              TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_dashboard_aggregation_enabled
ON dashboard_aggregation_definition
(
    enabled,
    display_order
);


CREATE TABLE dashboard_result
(
    dashboard_result_id     BIGSERIAL PRIMARY KEY,

    import_job_id           BIGINT NOT NULL,

    aggregation_code        VARCHAR(100) NOT NULL,

    category                VARCHAR(200),

    series                  VARCHAR(200),

    sub_category            VARCHAR(200),

    metric_value            NUMERIC,

    secondary_value         NUMERIC,

    tertiary_value          NUMERIC,

    sort_order              INTEGER,

    payload                 JSONB,

    calculated_at           TIMESTAMP NOT NULL
);

CREATE INDEX idx_dashboard_result_import
ON dashboard_result
(
    import_job_id,
    aggregation_code
);

CREATE INDEX idx_dashboard_result_latest
ON dashboard_result
(
    calculated_at DESC
);

    $sql$

        COALESCE(
            NULLIF(TRIM(application_service_name), ''),
