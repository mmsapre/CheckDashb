-- ============================================================================
-- SEVERITY DISTRIBUTION
-- ============================================================================

INSERT INTO dashboard_aggregation_definition
(
    aggregation_code,
    aggregation_name,
    chart_type,
    display_order,
    enabled,
    description,
    aggregation_sql
)
VALUES
(
    'SEVERITY_DISTRIBUTION',
    'Severity Distribution',
    'DONUT',
    15,
    TRUE,
    'Distribution of vulnerabilities by severity',
$$
SELECT
    :importJobId                                   AS import_job_id,
    'SEVERITY_DISTRIBUTION'                        AS aggregation_code,
    severity                                       AS category,
    severity                                       AS series,
    NULL                                           AS sub_category,
    COUNT(*)                                       AS metric_value,
    ROUND(
        COUNT(*) * 100.0 /
        SUM(COUNT(*)) OVER (),
        2
    )                                              AS secondary_value,
    NULL                                           AS tertiary_value,
    DENSE_RANK() OVER
    (
        ORDER BY
            CASE severity
                WHEN 'CRITICAL' THEN 1
                WHEN 'HIGH' THEN 2
                WHEN 'MEDIUM' THEN 3
                WHEN 'LOW' THEN 4
                WHEN 'INFO' THEN 5
                ELSE 99
            END
    )                                              AS sort_order,
    jsonb_build_object(
        'count', COUNT(*),
        'percentage',
        ROUND(
            COUNT(*) * 100.0 /
            SUM(COUNT(*)) OVER (),
            2
        )
    )                                              AS payload,
    CURRENT_TIMESTAMP                              AS calculated_at
FROM vulnerability_finding
WHERE import_job_id = :importJobId
GROUP BY severity
$$
);
