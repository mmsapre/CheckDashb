import client from "./ApiClient";

import type {
    DashboardFilter,
    ExecutiveDashboardResponse
} from "../model/dashboard";

export async function getLatestExecutiveDashboard(
    filter: DashboardFilter
): Promise<ExecutiveDashboardResponse> {

    const response =
        await client.post<ExecutiveDashboardResponse>(
            "/dashboard/latest/executive",
            filter
        );

    return response.data;
}

export async function getExecutiveDashboard(
    importJobId: number,
    filter: DashboardFilter
): Promise<ExecutiveDashboardResponse> {

    const response =
        await client.post<ExecutiveDashboardResponse>(
            `/dashboard/imports/${importJobId}/executive`,
            filter
        );

    return response.data;
}
